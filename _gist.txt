/* ===================== GitHub Gist 云端同步（照搬工作台方案） ===================== */
var SYNC_KEY='lstm_sync_v1', SYNC_FILE='lstm_data.json';
var sync={enabled:false,token:'',gistId:'',_t:null,_retry:null,_lastOk:0};
try{var _s=JSON.parse(localStorage.getItem(SYNC_KEY)||'{}');sync.enabled=!!_s.enabled;sync.token=_s.token||'';sync.gistId=_s.gistId||'';}catch(e){}
function saveSyncCfg(){localStorage.setItem(SYNC_KEY,JSON.stringify({enabled:sync.enabled,token:sync.token,gistId:sync.gistId}));}
function syncStatus(msg,cls){var el=document.getElementById('syncStatus');if(!el)return;el.textContent=msg||'';el.className='sync-status'+(cls?' '+cls:'');}
function syncPayload(){return JSON.stringify({v:2,updatedAt:Date.now(),goals:state.goals,records:state.records,weeklyNotes:state.weeklyNotes,journal:state.journal});}
async function gistReq(method,id,body){
  var url=id?('https://api.github.com/gists/'+id):'https://api.github.com/gists';
  var opt={method:method,headers:{'Authorization':'Bearer '+sync.token,'Content-Type':'application/json','Accept':'application/vnd.github+json'}};
  if(body)opt.body=JSON.stringify(body);
  var r=await fetch(url,opt);
  if(r.status===401)throw new Error('GitHub 401 · Token 已失效，请重新生成带 gist 权限的新 Token');
  if(r.status===403)throw new Error('GitHub 403 · Token 缺少 gist 权限，请重新生成时勾选 gist');
  if(!r.ok)throw new Error('GitHub '+r.status);
  return r.json();
}
async function pushSync(){
  if(!sync.enabled||!sync.token)return;
  try{
    var body=syncPayload();
    if(!sync.gistId){var g=await gistReq('POST',null,{description:'学习目标管理台数据同步',public:false,files:{[SYNC_FILE]:{content:body}}});sync.gistId=g.id;saveSyncCfg();}
    else{await gistReq('PATCH',sync.gistId,{files:{[SYNC_FILE]:{content:body}}});}
    showGistInfo();
    syncStatus('已同步 '+new Date().toLocaleTimeString(),'ok');
    sync._lastOk=Date.now();
    if(sync._retry){clearInterval(sync._retry);sync._retry=null;}
  }catch(e){
    if(/40[13]/.test(e&&e.message||'')){if(sync._retry){clearInterval(sync._retry);sync._retry=null;}syncStatus('⚠️ '+(e.message||'同步失败')+'（请更新 Token）','err');return;}
    var off=(typeof navigator!=='undefined'&&navigator.onLine===false);
    syncStatus(off?'离线，改动将在联网后上传':'同步失败，将自动重试：'+(e&&e.message?e.message:e),'warn');
    startPendingRetry();
  }
}
function startPendingRetry(){
  if(sync._retry)return;
  sync._retry=setInterval(function(){if(!sync.enabled||!sync.token){clearInterval(sync._retry);sync._retry=null;return;}pushSync();},6000);
}
function showGistInfo(){
  var el=document.getElementById('syncGistInfo');if(!el)return;
  el.innerHTML=sync.gistId?('已绑定私有 Gist：<a href="https://gist.github.com/'+sync.gistId+'" target="_blank" rel="noopener">'+sync.gistId+'</a><br>在两台设备用<b>相同 Token + 此 ID</b> 即可共享同一份数据。'):'';
}
function schedulePush(){if(!sync.enabled||!sync.token)return;clearTimeout(sync._t);sync._t=setTimeout(pushSync,800);}
function mergeById(local,incoming,key){
  key=key||'id';
  var map={};
  (local||[]).forEach(function(x){if(x&&x[key]!=null)map[x[key]]=x;});
  (incoming||[]).forEach(function(x){if(x&&x[key]!=null){if(map[x[key]])Object.assign(map[x[key]],x);else map[x[key]]=x;}});
  return Object.keys(map).map(function(k){return map[k];});
}
async function pullSync(silent){
  if(!sync.enabled||!sync.token||!sync.gistId)return;
  try{
    var g=await gistReq('GET',sync.gistId);
    var f=g.files&&g.files[SYNC_FILE];if(!f||!f.content)return;
    var d=JSON.parse(f.content);
    var inc=d.state||d;
    var before=JSON.stringify(state);
    if(Array.isArray(inc.goals))state.goals=mergeById(state.goals,inc.goals);
    if(Array.isArray(inc.records))state.records=mergeById(state.records,inc.records);
    if(inc.weeklyNotes)state.weeklyNotes=Object.assign({},state.weeklyNotes||{},inc.weeklyNotes);
    if(Array.isArray(inc.journal))state.journal=mergeById(state.journal,inc.journal);
    var after=JSON.stringify(state);
    if(before!==after){
      await saveDataAsync();
      var ae=document.activeElement;
      var typing=ae&&(ae.tagName==='INPUT'||ae.tagName==='TEXTAREA');
      var modalOpen=document.querySelector('.modal-mask.show,.sheet-mask.show');
      if(!typing&&!modalOpen)renderAll();else renderMine();
    }
    sync._lastOk=Date.now();
    if(!silent)syncStatus('已拉取 '+new Date().toLocaleTimeString(),'ok');
  }catch(e){
    if(/40[13]/.test(e&&e.message||'')){if(sync._retry){clearInterval(sync._retry);sync._retry=null;}syncStatus('⚠️ '+(e.message||'拉取失败')+'（请更新 Token）','err');return;}
    if(!silent)syncStatus('拉取失败：'+(e&&e.message?e.message:e),'err');
  }
}
async function syncNow(){if(!sync.enabled||!sync.token){syncStatus('请先启用同步','err');return;}syncStatus('同步中…');await pullSync();await pushSync();}
var _autoPull=null;
function startAutoPull(){if(_autoPull)return;_autoPull=setInterval(function(){if(typeof navigator!=='undefined'&&navigator.onLine===false)return;pullSync(true);},45000);}
function syncEnable(){
  var tok=document.getElementById('syncToken').value.trim();
  var gid=document.getElementById('syncGistId').value.trim();
  if(!tok){toast('请输入 Token','err');return;}
  sync.token=tok;sync.gistId=gid;sync.enabled=true;saveSyncCfg();
  pushSync().then(function(){showConnectedUI();startAutoPull();toast('已启用并同步 ✅','ok');}).catch(function(e){toast(e.message,'err');});
}
function syncDisable(){
  if(!confirm('断开后数据仍保留在本地和 Gist 中。确定断开？'))return;
  sync.enabled=false;sync.token='';sync.gistId='';saveSyncCfg();
  showFormUI();toast('已断开云端同步','ok');
}
function showFormUI(){
  var f=document.getElementById('ghForm'),c=document.getElementById('ghConnected');
  if(f)f.style.display='';if(c)c.style.display='none';
  var d=document.getElementById('syncStatus');if(d)d.textContent='';
}
function showConnectedUI(){
  var f=document.getElementById('ghForm'),c=document.getElementById('ghConnected');
  if(f)f.style.display='none';if(c)c.style.display='';
  showGistInfo();renderMine();
}
function syncInit(){
  if(sync.enabled&&sync.token){
    if(sync.gistId)showConnectedUI();else showFormUI();
    startAutoPull();
    pullSync(true);
  }
}
function syncDiag(){
  var out=document.getElementById('ghDiagOut');if(!out)return;out.style.display='';out.textContent='诊断中…';
  var lines=['== Gist 同步诊断 =='];
  if(!sync.enabled||!sync.token){lines.push('✗ 未启用：请填入 Token（带 gist 权限）+ 可选 Gist ID，点「启用并同步」');out.textContent=lines.join('\\n');return;}
  lines.push('✓ 已配置 Token('+sync.token.length+'字符) · GistID='+(sync.gistId||'(空，将自动创建)'));
  lines.push('最近成功同步：'+(sync._lastOk?new Date(sync._lastOk).toLocaleString():'从未'));
  lines.push('— 联网后浏览器会向 api.github.com/gists 发请求（控制台 Network 可见）');
  out.textContent=lines.join('\\n');
}
